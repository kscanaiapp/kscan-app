const dns=require('node:dns').promises,fs=require('fs');
(async()=>{const records=[];for(const hostname of ['127.0.0.1.nip.io','169.254.169.254.nip.io','127.0.0.1.sslip.io']){try{records.push({hostname,answers:await dns.lookup(hostname,{all:true}),method:'DNS lookup only; no HTTPS/private-address request'});}catch(e){records.push({hostname,errorCode:e.code,method:'DNS lookup only; no HTTPS/private-address request'});}}fs.writeFileSync('C:/b35readback1030/media-dns-readonly.json',JSON.stringify({capturedAt:new Date().toISOString(),records,limitation:'Local resolver evidence only. No connection to resolved addresses; does not prove Supabase DNS/TLS/egress behavior or private resource access.'},null,2)+'\n');console.log(JSON.stringify(records));})().catch(e=>{console.error(e.code);process.exitCode=1;});
